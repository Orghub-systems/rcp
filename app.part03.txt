  function dialog(title, body, saveLabel = 'Zapisz') {
    const wrap = document.createElement('div');
    wrap.className = 'dialog-backdrop';
    wrap.innerHTML = `<div class="dialog"><h3>${esc(title)}</h3><div>${body}</div><div class="dialog-actions"><button class="btn btn-light" data-cancel>Anuluj</button><button class="btn btn-dark" data-save>${esc(saveLabel)}</button></div></div>`;
    document.body.appendChild(wrap);
    wrap.querySelector('[data-cancel]').addEventListener('click', () => wrap.remove());
    wrap.addEventListener('click', e => { if (e.target === wrap) wrap.remove(); });
    return { wrap, save: wrap.querySelector('[data-save]') };
  }

  function slugify(value) {
    return String(value || '')
      .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
      .toLowerCase().trim()
      .replace(/[^a-z0-9]+/g, '-')
      .replace(/^-+|-+$/g, '');
  }

  function openAddOrganization() {
    const { wrap, save } = dialog('Dodaj firmę', `<div class="form-grid">
      <div class="field"><label>Nazwa firmy</label><input id="orgName" class="input" required></div>
      <div class="field"><label>Identyfikator</label><input id="orgSlug" class="input" required placeholder="np. intevero"></div>
      <div class="field span-2"><label>Strefa czasowa</label><input id="orgTimezone" class="input" value="Europe/Warsaw" required></div>
      <div class="field"><label>Imię administratora</label><input id="orgAdminFirst" class="input" required></div>
      <div class="field"><label>Nazwisko administratora</label><input id="orgAdminLast" class="input"></div>
      <div class="field span-2"><label>E-mail administratora</label><input id="orgAdminEmail" class="input" type="email" required></div>
    </div>`, 'Utwórz firmę');

    const nameInput = wrap.querySelector('#orgName');
    const slugInput = wrap.querySelector('#orgSlug');
    let slugTouched = false;
    slugInput.addEventListener('input', () => { slugTouched = true; });
    nameInput.addEventListener('input', () => { if (!slugTouched) slugInput.value = slugify(nameInput.value); });

    save.addEventListener('click', async () => {
      const p_name = nameInput.value.trim();
      const p_slug = slugInput.value.trim().toLowerCase();
      const p_timezone = wrap.querySelector('#orgTimezone').value.trim();
      const p_admin_first_name = wrap.querySelector('#orgAdminFirst').value.trim();
      const p_admin_last_name = wrap.querySelector('#orgAdminLast').value.trim();
      const p_admin_email = wrap.querySelector('#orgAdminEmail').value.trim().toLowerCase();
      if (!p_name || !p_slug || !p_admin_first_name || !p_admin_email) return toast('Uzupełnij nazwę, identyfikator i dane administratora.', 'error');
      save.disabled = true;
      const { error } = await db.rpc('platform_create_organization', { p_name, p_slug, p_admin_first_name, p_admin_last_name, p_admin_email, p_timezone });
      if (error) { toast(error.message, 'error'); save.disabled = false; return; }
      wrap.remove();
      toast('Firma utworzona.', 'success');
      await loadIdentity();
      state.view = 'platform';
      localStorage.setItem('rcp:view', 'platform');
      await loadPlatformData();
      renderPlatform();
    });
  }

  function openAddEmployee() {
    const { wrap, save } = dialog('Dodaj pracownika', `<div class="form-grid">
      <div class="field"><label>Imię</label><input id="empFirst" class="input" required></div>
      <div class="field"><label>Nazwisko</label><input id="empLast" class="input"></div>
      <div class="field span-2"><label>E-mail logowania</label><input id="empEmail" class="input" type="email" required placeholder="pracownik@firma.pl"></div>
    </div>`, 'Dodaj');
    save.addEventListener('click', async () => {
      const first_name = wrap.querySelector('#empFirst').value.trim();
      const last_name = wrap.querySelector('#empLast').value.trim() || null;
      const login_email = wrap.querySelector('#empEmail').value.trim().toLowerCase();
      if (!first_name || !login_email) return toast('Podaj imię i e-mail.', 'error');
      save.disabled = true;
      const { error } = await db.from('organization_members').insert({ organization_id: state.membership.organization_id, first_name, last_name, login_email, role: 'employee', active: true });
      if (error) { toast(error.message, 'error'); save.disabled = false; return; }
      wrap.remove(); toast('Pracownik dodany.', 'success'); await loadAdminData(); renderAdmin();
    });
  }
