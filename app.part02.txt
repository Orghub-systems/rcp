find(m => m.id === memberId);
    if (!membership) return;
    state.membership = membership;
    state.organization = state.organizations.find(o => o.id === membership.organization_id) || null;
    state.view = 'organization';
    state.adminTab = 'dashboard';
    localStorage.setItem('rcp:selected-membership', memberId);
    localStorage.setItem('rcp:view', 'organization');
    loading();
    await renderSelectedTenant();
  }

  async function loadEmployeeData() {
    const memberId = state.membership.id;
    const { data: sessions, error } = await db.from('work_sessions_with_earnings')
      .select('*')
      .eq('member_id', memberId)
      .gte('started_at', thisMonthStartIso())
      .order('started_at', { ascending: false });
    if (error) throw error;
    state.sessions = sessions || [];
    state.currentSession = state.sessions.find(s => !s.ended_at) || null;
  }

  function employeeTotals() {
    const today = localDateKey(new Date());
    let todayMinutes = 0;
    let monthMinutes = 0;
    let earnings = 0;
    for (const s of state.sessions) {
      const mins = durationMinutes(s.started_at, s.ended_at || Date.now());
      monthMinutes += mins;
      if (localDateKey(new Date(s.started_at)) === today) todayMinutes += mins;
      if (s.hourly_rate != null) earnings += (mins / 60) * Number(s.hourly_rate);
    }
    return { todayMinutes, monthMinutes, earnings };
  }

  async function togglePunch() {
    if (state.busy) return;
    state.busy = true;
    const btn = document.getElementById('punchBtn');
    if (btn) btn.disabled = true;
    const fn = state.currentSession ? 'stop_work' : 'start_work';
    const { error } = await db.rpc(fn, { p_member_id: state.membership.id });
    if (error) {
      toast(error.message, 'error');
      state.busy = false;
      if (btn) btn.disabled = false;
      return;
    }
    toast(state.currentSession ? 'Czas pracy zakończony.' : 'Czas pracy rozpoczęty.', 'success');
    await loadEmployeeData();
    state.busy = false;
    renderEmployee();
  }

  async function loadAdminData() {
    const orgId = state.membership.organization_id;
    const monthStart = thisMonthStartIso();
    const [membersRes, sessionsRes] = await Promise.all([
      db.from('organization_members').select('*').eq('organization_id', orgId).order('active', { ascending: false }).order('first_name'),
      db.from('work_sessions_with_earnings').select('*').eq('organization_id', orgId).gte('started_at', monthStart).order('started_at', { ascending: false })
    ]);
    if (membersRes.error) throw membersRes.error;
    if (sessionsRes.error) throw sessionsRes.error;
    state.members = membersRes.data || [];
    state.adminSessions = sessionsRes.data || [];

    const ids = state.members.map(m => m.id);
    if (ids.length) {
      const ratesRes = await db.from('employee_rates').select('*').in('member_id', ids).order('valid_from', { ascending: false });
      if (ratesRes.error) throw ratesRes.error;
      state.rates = ratesRes.data || [];
    } else {
      state.rates = [];
    }
  }

  function adminMetrics() {
    const employees = state.members.filter(m => m.role === 'employee' && m.active);
    const open = state.adminSessions.filter(s => !s.ended_at);
    const today = localDateKey(new Date());
    let todayMinutes = 0;
    let monthEarnings = 0;
    for (const s of state.adminSessions) {
      const mins = durationMinutes(s.started_at, s.ended_at || Date.now());
      if (localDateKey(new Date(s.started_at)) === today) todayMinutes += mins;
      if (s.hourly_rate != null) monthEarnings += (mins / 60) * Number(s.hourly_rate);
    }
    return { employees: employees.length, open: open.length, todayMinutes, monthEarnings };
  }

  function memberCurrent(memberId) {
    return state.adminSessions.find(s => s.member_id === memberId && !s.ended_at) || null;
  }

  function memberMonth(memberId) {
    const list = state.adminSessions.filter(s => s.member_id === memberId);
    return list.reduce((acc, s) => {
      const mins = durationMinutes(s.started_at, s.ended_at || Date.now());
      acc.minutes += mins;
      if (s.hourly_rate != null) acc.earnings += mins / 60 * Number(s.hourly_rate);
      return acc;
    }, { minutes: 0, earnings: 0 });
  }
